async function initJukebox() { let targetPlaylists = []; let modeTitle = isMGMode ? "Loading Radio M.Gorille" : "Loading Your Playlists"; document.getElementById('current-song-title').innerText = "Loading..."; if (isMGMode) { targetPlaylists = mgPlaylists; localStorage.setItem('saved_mg_mode', 'true'); } else { const rawText = document.getElementById('playlist-input').value; localStorage.setItem('saved_jukebox_playlists', rawText); localStorage.setItem('saved_mg_mode', 'false'); targetPlaylists = [...new Set(rawText.split(/[\n,]+/).map(i => { if (i.includes('list=')) { try { const url = new URL(i.startsWith('http') ? i : 'https://www.youtube.com' + i); return url.searchParams.get('list'); } catch (e) { return null; } } return i.trim().length > 5 ? i.trim() : null; }).filter(Boolean))]; } if (targetPlaylists.length === 0) { alert('Please enter at least one valid playlist URL.'); document.getElementById('current-song-title').innerText = "Waiting for playback to begin..."; return; } isJukeboxActive = true; masterTrackPool = []; if (player && typeof player.mute === 'function') { player.mute(); } document.getElementById('tutorial-overlay').classList.add('hidden'); const overlay = document.getElementById('loading-overlay'); const fillBar = document.getElementById('progress-bar-fill'); const statusLabel = document.getElementById('overlay-status'); overlay.classList.remove('hidden'); fillBar.style.width = '0%'; if (isMGMode) { statusLabel.innerText = `${modeTitle}...`; } else { statusLabel.innerText = `${modeTitle}... (0/${targetPlaylists.length})`; } document.getElementById('status-display').innerText = `${modeTitle}...`; for (let i = 0; i < targetPlaylists.length; i++) { let currentCount = i + 1; let percent = Math.round((currentCount / targetPlaylists.length) * 100); fillBar.style.width = percent + '%'; if (isMGMode) { statusLabel.innerText = `${modeTitle}...`; } else { statusLabel.innerText = `${modeTitle}... (${currentCount}/${targetPlaylists.length})`; } player.loadPlaylist({list: targetPlaylists[i], listType: 'playlist', index: 0}); // Safety-capped timeout loop: will never hang infinitely let fetchedTracks = false; let startTime = Date.now(); while (Date.now() - startTime < 2500) { await new Promise(r => setTimeout(r, 300)); try { const tracks = player.getPlaylist(); if (tracks && Array.isArray(tracks) && tracks.length > 0) { tracks.forEach(vId => masterTrackPool.push(vId)); fetchedTracks = true; break; } } catch (e) {} } // Fallback: If network/adblocker blocks getPlaylist(), use the playlist ID directly if (!fetchedTracks) { masterTrackPool.push(targetPlaylists[i]); } } masterTrackPool = [...new Set(masterTrackPool)]; if (player && typeof player.unMute === 'function') { player.unMute(); } overlay.classList.add('hidden'); const panel = document.getElementById('control-panel'); const hint = document.getElementById('click-hint'); if (!panel.classList.contains('collapsed')) { panel.classList.add('collapsed'); hint.style.display = 'inline'; } if (masterTrackPool.length > 0) { playRandomMasterTrack(); } else { alert('Could not index tracks. Make sure the playlist links are public.'); document.getElementById('current-song-title').innerText = "Waiting for playback to begin..."; } }
